	MAFIO_voidSetEXTIConfiguration(0 , AFIO_PB);
	MEXTI_voidEnableEXTI(0);

	MEXTI_voidSetSignalLatch(0 ,  FALING_EDGE);


	/* EXTI1 -> PB1 */

	MAFIO_voidSetEXTIConfiguration(1 , AFIO_PB);

	MEXTI_voidEnableEXTI(1);

	MEXTI_voidSetSignalLatch(1 ,  FALING_EDGE);



	/* ================= NVIC ================= */

	MNVIC_voidEnableInterrupt(6); /* EXTI0 */

	MNVIC_voidEnableInterrupt(7); /* EXTI1 */



	/* ================= while ================= */

	while(1)
	{
		for(int i = 0; i < 70; i++)
		{
			display_time(hours, minutes, seconds);
		}

		if(running == 1)
		{
			seconds++;

			if(seconds > 9)
			{
				seconds = 0;
				minutes++;
			}

			if(minutes > 9)
			{
				minutes = 0;
				hours++;
			}

			if(hours > 9)
			{
				hours = 0;
			}

			MSTK_voidDelayMs(1000);
		}
	}
}

int main(void)
{
	/* ================= RCC ================= */

	RCC_voidInitSysClock();

	/* Enable GPIOA Clock */
	RCC_voidEableClock(RCC_APB2 , 2);

	/* Enable GPIOB Clock */
	RCC_voidEableClock(RCC_APB2 , 3);

	/* Enable AFIO Clock */
	RCC_voidEableClock(RCC_APB2 , 0);


	/* ================= EXTI ================= */

	MEXTI_voidInit();


	/* ================= SysTick ================= */

	MSTK_voidInit();


	/* ================= NVIC ================= */

	MNVIC_voidInit();


	/* ================= GPIO ================= */

	/* Segments PA0 -> PA6 */

	for(int i = 0 ; i < 7 ; i++)
	{
		MGPIO_voidSetpinDirection(GPIOA , i , OUTPUT_SPEED_10MHZ_PP);
	}


	/* Digits */

	MGPIO_voidSetpinDirection(GPIOB , PIN12 , OUTPUT_SPEED_10MHZ_PP);
	MGPIO_voidSetpinDirection(GPIOB , PIN13 , OUTPUT_SPEED_10MHZ_PP);
	MGPIO_voidSetpinDirection(GPIOB , PIN14 , OUTPUT_SPEED_10MHZ_PP);


	/* LEDs */

	MGPIO_voidSetpinDirection(GPIOA , PIN8 , OUTPUT_SPEED_10MHZ_PP);
	MGPIO_voidSetpinDirection(GPIOA , PIN9 , OUTPUT_SPEED_10MHZ_PP);


	/* Switches PB0 PB1 */

	MGPIO_voidSetpinDirection(GPIOB , PIN0 , INPUT_PULL_UP_DOWN);
	MGPIO_voidSetpinDirection(GPIOB , PIN1 , INPUT_PULL_UP_DOWN);


	/* Pull Up */

	MGPIO_voidSetpinValue(GPIOB , PIN0 , GPIO_HIGH);
	MGPIO_voidSetpinValue(GPIOB , PIN1 , GPIO_HIGH);


	/* ================= AFIO + EXTI ================= */

	/* EXTI0 -> PB0 */

	MAFIO_voidSetEXTIConfiguration(0 , AFIO_PB);

	MEXTI_voidEnableEXTI(0);

	MEXTI_voidSetSignalLatch(0 , FALLING_EDGE);


	/* EXTI1 -> PB1 */

	MAFIO_voidSetEXTIConfiguration(1 , AFIO_PB);

	MEXTI_voidEnableEXTI(1);

	MEXTI_voidSetSignalLatch(1 , FALLING_EDGE);


	/* ================= NVIC ================= */

	MNVIC_voidEnableInterrupt(6);

	MNVIC_voidEnableInterrupt(7);


	/* ================= While ================= */

	while(1)
	{
		for(int i = 0 ; i < 70 ; i++)
		{
			display_time(hours , minutes , seconds);
		}


		if(running == 1)
		{
			seconds++;

			if(seconds > 9)
			{
				seconds = 0;

				minutes++;
			}


			if(minutes > 9)
			{
				minutes = 0;

				hours++;
			}


			if(hours > 9)
			{
				hours = 0;
			}


		//	MSTK_voidDelayMs(1000);
		}
	}
}
